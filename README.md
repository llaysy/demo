МОДУЛЬ № 1
Разработка, администрирование и защита баз данных
ЭТАП 1 — Подготовка папки и создание БД
 
1
Создать папку для базы данных
Откройте Терминал (Ctrl+Alt+T) и выполните команды:
sudo mkdir /db
sudo chown reddatabase /db
⚠️ Важно: Папка /db должна принадлежать пользователю reddatabase — без этого Firebird не сможет создать файл БД.
 
2
Создать базу данных в РБД Эксперт
Запустите РБД Эксперт. Нажмите «Создать базу данных». Заполните поля:
Параметры новой БД
Имя подключения
DemoConnection
Файл базы данных
/db/demo.fdb
Имя пользователя
sysdba
Пароль
пароль администратора (задан при установке)
Кодировка
UTF8
Размер страницы
8192
Нажмите «Create». В дереве появится DemoConnection.
 
3
Подключиться к БД
ПКМ по DemoConnection → Connect. После подключения в дереве появятся узлы Tables, Views и др.
✅ Проверка: В дереве под DemoConnection должны быть узлы Tables, Sequences, Functions и т.д.
 
ЭТАП 2 — Создание таблиц
 
Откройте Query Editor (меню Database → Query Editor). Создавайте таблицы строго по одной в порядке ниже: выделите код одной таблицы → F5.
⚠️ Важно: Сначала — справочники (без FK), затем — зависимые таблицы. Нарушение порядка даст ошибку FK.
 
1. CATEGORY
CREATE TABLE CATEGORY
(
   CATEGORYID   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   CATEGORYNAME VARCHAR(50)
 );
 
2. MANUFACTURER
CREATE TABLE MANUFACTURER
(
   MANUFACTURERID   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   MANUFACTURERNAME VARCHAR(50)
 );
 
3. SUPPLIER
CREATE TABLE SUPPLIER
(
   SUPPLIERID   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   SUPPLIERNAME VARCHAR(50)
 );
 
4. PRODUCT
CREATE TABLE PRODUCT
(
   PRODUCTARTICUL VARCHAR(10) PRIMARY KEY,
   PRODUCTNAME    VARCHAR(50),
   PRODUCTUNIT    VARCHAR(10),
   PRODUCTPRICE   DECIMAL(10,2),
   SUPPLIER       INT, FOREIGN KEY (SUPPLIER)    REFERENCES SUPPLIER(SUPPLIERID),
   MANUFACTURER   INT, FOREIGN KEY (MANUFACTURER) REFERENCES MANUFACTURER(MANUFACTURERID),
   CATEGORY       INT, FOREIGN KEY (CATEGORY)    REFERENCES CATEGORY(CATEGORYID),
   DISCOUNT       INT,
   COUNTINSTOCK   INT,
   DESCRIPTION    VARCHAR(200),
   PHOTOPATH      VARCHAR(200)
 );
 
5. ROLE
CREATE TABLE ROLE
(
   ROLEID   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   ROLENAME VARCHAR(50)
 );
 
6. "USER"
CREATE TABLE "USER"
(
   USERID     INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   ROLE       INT, FOREIGN KEY (ROLE) REFERENCES ROLE(ROLEID),
   LOGIN      VARCHAR(50),
   PASSWORD   VARCHAR(50),
   FIRSTNAME  VARCHAR(50),
   LASTNAME   VARCHAR(50),
   MIDDLENAME VARCHAR(50)
 );
 
7. STATUS
CREATE TABLE STATUS
(
   STATUSID   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   STATUSNAME VARCHAR(50)
 );
 
8. ADDRESS
CREATE TABLE ADDRESS
(
   ADDRESSID   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   ADDRESSNAME VARCHAR(200)
 );
 
9. "ORDER"
CREATE TABLE "ORDER"
(
   ORDERID        INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   ORDERSTATUS    INT, FOREIGN KEY (ORDERSTATUS)  REFERENCES STATUS(STATUSID)   ON DELETE CASCADE ON UPDATE CASCADE,
   ORDERADDRESS   INT, FOREIGN KEY (ORDERADDRESS) REFERENCES ADDRESS(ADDRESSID)  ON DELETE CASCADE ON UPDATE CASCADE,
   ORDERDATE      DATE,
   ORDERDATEISSUE DATE,
   ORDERUSER      INT, FOREIGN KEY (ORDERUSER)   REFERENCES "USER"(USERID)     ON DELETE CASCADE ON UPDATE CASCADE,
   ORDERCODE      INT
 );
 
10. ORDERPRODUCT
CREATE TABLE ORDERPRODUCT
(
   ORDERPRODUCTID    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
   "ORDER"           INT, FOREIGN KEY ("ORDER") REFERENCES "ORDER"(ORDERID)            ON DELETE CASCADE ON UPDATE CASCADE,
   PRODUCT           VARCHAR(10), FOREIGN KEY (PRODUCT) REFERENCES PRODUCT(PRODUCTARTICUL) ON DELETE CASCADE ON UPDATE CASCADE,
   ORDERPRODUCTCOUNT INT
 );
 
✅ Проверка: После каждого F5 нижняя панель должна показать «Execute successful» без ошибок. В дереве Tables появятся новые таблицы.

Шаг 5 — Импорт заказов (ORDER + ORDERPRODUCT)
Файл: Заказ_import.xlsx. Структура: Номер заказа | Артикул заказа | Дата заказа | Дата доставки | Адрес (ID) | ФИО клиента | Код | Статус
⚠️ Даты: В файле даты хранятся как числа Excel (напр. 45349 = 2024-02-12). LibreOffice/Excel покажет их правильно если поставить формат «Дата». Перед импортом убедитесь что столбцы дат имеют формат даты, а не число.
⚠️ Артикул заказа: Поле «Артикул заказа» содержит несколько товаров через запятую (напр. «А112Т4, 2, G843H5, 2» = товар А112Т4 × 2 шт. и G843H5 × 2 шт.). Это нужно разбить на строки ORDERPRODUCT. Сначала импортируйте ORDER, потом вручную создайте записи ORDERPRODUCT.
Алгоритм работы с заказами:
7. Импортировать заказы в ORDER (без поля артикулов): номер, дата, дата доставки, ORDERADDRESS (ID адреса), ORDERUSER (ID пользователя), ORDERCODE, ORDERSTATUS (ID статуса).
8. Для каждого заказа разбить поле «Артикул заказа» и вставить строки в ORDERPRODUCT через SQL.
 
Пример SQL для ORDERPRODUCT (заказ 1: А112Т4 × 2, G843H5 × 2, ORDERID=1):
INSERT INTO ORDERPRODUCT ("ORDER", PRODUCT, ORDERPRODUCTCOUNT) VALUES (1, 'А112Т4', 2);
INSERT INTO ORDERPRODUCT ("ORDER", PRODUCT, ORDERPRODUCTCOUNT) VALUES (1, 'G843H5', 2);

ЭТАП 3 — Подключение к базе данных
 
Создайте файл Services/DatabaseService.cs:
using FirebirdSql.Data.FirebirdClient;
 
public static class DatabaseService
{
   private const string ConnectionString =
       "DataSource=localhost;Database=/db/demo.fdb;" +
       "User=sysdba;Password=masterkey;Charset=UTF8;";
 
   public static FbConnection GetConnection()
   {
       var conn = new FbConnection(ConnectionString);
       conn.Open();
       return conn;
   }
}
⚠️ Пароль: Замените masterkey на реальный пароль sysdba вашей установки.

private void OnLoginClick(object? sender, RoutedEventArgs e)
{
   string login = LoginBox.Text ?? "";
   string pass  = PassBox.Text  ?? "";
 
   using var conn = DatabaseService.GetConnection();
   string sql = "SELECT u.USERID, r.ROLENAME, u.FIRSTNAME, u.LASTNAME " +
                'FROM "USER" u JOIN ROLE r ON u.ROLE=r.ROLEID ' +
                "WHERE u.LOGIN=@l AND u.PASSWORD=@p";
   using var cmd = new FbCommand(sql, conn);
   cmd.Parameters.AddWithValue("@l", login);
   cmd.Parameters.AddWithValue("@p", pass);
   using var reader = cmd.ExecuteReader();
   if (reader.Read())
   {
       int    userId   = reader.GetInt32(0);
       string roleName = reader.GetString(1);
       // Открыть главное окно с нужной ролью
       var mainWin = new ProductsWindow(userId, roleName);
       mainWin.Show();
       this.Close();
   }
   else
   {
       // Показать ошибку
       ErrorText.IsVisible = true;
   }
}

// Модель товара
public class ProductViewModel
{
   public string Articul      { get; set; } = "";
   public string Name         { get; set; } = "";
   public decimal Price       { get; set; }
   public int    Discount     { get; set; }
   public int    CountInStock { get; set; }
   public string Category     { get; set; } = "";
   public string PhotoPath    { get; set; } = "";
   // Цвет фона: если скидка > 25 — #23E1EF, иначе прозрачный
   public string RowColor => Discount > 25 ? "#23E1EF" : "Transparent";
}
 
// Загрузка из БД
private void LoadProducts(string search="", string category="", string sortCol="", bool asc=true)
{
   var list = new List<ProductViewModel>();
   using var conn = DatabaseService.GetConnection();
   string sql = "SELECT p.PRODUCTARTICUL, p.PRODUCTNAME, p.PRODUCTPRICE," +
                " p.DISCOUNT, p.COUNTINSTOCK, c.CATEGORYNAME, p.PHOTOPATH" +
                " FROM PRODUCT p" +
                " LEFT JOIN CATEGORY c ON p.CATEGORY = c.CATEGORYID" +
                " WHERE (UPPER(p.PRODUCTNAME) LIKE UPPER(@s))";
   if (!string.IsNullOrEmpty(category))
       sql += " AND c.CATEGORYNAME = @cat";
   sql += $" ORDER BY {(string.IsNullOrEmpty(sortCol) ? "p.PRODUCTNAME" : sortCol)}";
   sql += asc ? " ASC" : " DESC";
 
   using var cmd = new FbCommand(sql, conn);
   cmd.Parameters.AddWithValue("@s", $"%{search}%");
   if (!string.IsNullOrEmpty(category))
       cmd.Parameters.AddWithValue("@cat", category);
 
   using var r = cmd.ExecuteReader();
   while (r.Read())
       list.Add(new ProductViewModel {
           Articul      = r.GetString(0),
           Name         = r.GetString(1),
           Price        = r.GetDecimal(2),
           Discount     = r.GetInt32(3),
           CountInStock = r.GetInt32(4),
           Category     = r.IsDBNull(5) ? "" : r.GetString(5),
           PhotoPath    = r.IsDBNull(6) ? "" : r.GetString(6)
       });
   ProductGrid.ItemsSource = list;
}
